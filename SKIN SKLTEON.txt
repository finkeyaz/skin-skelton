local modeCouleur = "blanc" -- blanc | bleu | rouge | violet | rgb | galaxy

local COULEURS = {
    blanc  = Color3.fromRGB(255, 255, 255),
    bleu   = Color3.fromRGB(0, 130, 255),
    rouge  = Color3.fromRGB(255, 45, 45),
    violet = Color3.fromRGB(170, 0, 255),
}

local GALAXIE = {
    Color3.fromRGB(25, 25, 80),   -- bleu nuit
    Color3.fromRGB(70, 20, 140),  -- violet profond
    Color3.fromRGB(150, 40, 190), -- magenta spatial
    Color3.fromRGB(40, 40, 120),  -- indigo
}

local function getCouleur()
    if modeCouleur == "rgb" then
        return Color3.fromHSV((os.clock() * 0.25) % 1, 1, 1)
    elseif modeCouleur == "galaxy" then
        local x = (os.clock() * 0.12) % #GALAXIE
        local i = math.floor(x) + 1
        local j = (i % #GALAXIE) + 1
        return GALAXIE[i]:Lerp(GALAXIE[j], x % 1)
    end
    return COULEURS[modeCouleur] or COULEURS.blanc
end

-- v4.1 : ne JAMAIS masquer les outils (pioche, armes, ...)
-- sinon la pioche devient invisible dès que tu l'équipes
local function estDansUnTool(obj)
    return obj:IsA("Tool") or obj:FindFirstAncestorOfClass("Tool") ~= nil
end

local function cacherObjet(obj)
    if estDansUnTool(obj) then return end
    if obj:IsA("BasePart") then
        obj.LocalTransparencyModifier = 1
        obj.Transparency = 1
    elseif obj:IsA("Decal") then
        obj.LocalTransparencyModifier = 1
        obj.Transparency = 1
    elseif obj:IsA("Accessory") then
        local h = obj:FindFirstChild("Handle")
        if h then
            h.LocalTransparencyModifier = 1
            h.Transparency = 1
        end
    end
end

-- Si le jeu remet la transparence à zéro, on re-cache immédiatement
local function protegerPartie(part)
    if not part:IsA("BasePart") then return end
    part:GetPropertyChangedSignal("Transparency"):Connect(function()
        if part.Transparency < 1 and part.Parent then
            part.Transparency = 1
        end
    end)
end

-- Masque TOUT le personnage, même s'il est custom.
-- v4 : UNE SEULE fois par personnage -> plus de fuite de listeners
local persosMasques = {}

local function masquerPerso(character)
    if not character or not character.Parent then return end
    if persosMasques[character] then return end
    persosMasques[character] = true

    local dejaProteges = {}
    for _, obj in ipairs(character:GetDescendants()) do
        if not estDansUnTool(obj) then
            cacherObjet(obj)
            if obj:IsA("BasePart") then
                protegerPartie(obj)
                dejaProteges[obj] = true
            end
        end
    end
    character.DescendantAdded:Connect(function(obj)
        if estDansUnTool(obj) then return end
        cacherObjet(obj)
        if obj:IsA("BasePart") and not dejaProteges[obj] then
            dejaProteges[obj] = true
            protegerPartie(obj)
        end
    end)
    character.AncestryChanged:Connect(function(_, parent)
        if not parent then
            persosMasques[character] = nil
        end
    end)

    print("[Skin] 🫥 Skin réel masqué pour " .. character.Name)
end

local function detruireClone(player)
    local c = clones[player]
    if c then
        if c.sync then c.sync:Disconnect() end
        if c.model then c.model:Destroy() end
        clones[player] = nil
    end
end

local function appliquerCouleur()
    local c = getCouleur()
    for _, cl in pairs(clones) do
        local model = cl.model
        if model and model.Parent then
            for _, s in ipairs(model:GetDescendants()) do
                if s:IsA("BasePart") then
                    s.Color = c
                end
            end
        end
    end
end

-- Racine de secours : HumanoidRootPart, sinon PrimaryPart,
-- sinon n'importe quelle BasePart du perso
local function trouverRacine(character)
    local r = character:FindFirstChild("HumanoidRootPart")
    if not (r and r:IsA("BasePart")) then
        r = character.PrimaryPart
    end
    if not (r and r:IsA("BasePart")) then
        r = character:FindFirstChildWhichIsA("BasePart", true)
    end
    return (r and r:IsA("BasePart")) and r or nil
end

-- Ancrage canonique : si le vrai perso est R6 (Left Arm, Torso, ...),
-- les racines de membres du squelette R15 s'y ancrent directement,
-- AVANT le "plus proche" qui peut se tromper de membre
local ANCRAGE_R6 = {
    UpperTorso    = "Torso",
    LowerTorso    = "Torso",
    Torso         = "Torso",
    LeftUpperArm  = "Left Arm",
    RightUpperArm = "Right Arm",
    LeftUpperLeg  = "Left Leg",
    RightUpperLeg = "Right Leg",
}

-- Association membre squelette -> partie réelle :
--   1) par nom exact -> copie directe de la partie réelle
--   2) sinon CHAÎNE -> suit son segment parent squelette
--      (main -> avant-bras -> bras -> torse, pied -> jambe)
--      => le squelette reste TOUJOURS connecté :
--         plus de mains / pieds flottants ou d'écart qui s'ouvre
--   3) racine de chaîne -> ancre R6 canonique, sinon partie réelle
--      la plus proche, avec un offset de POSITION UNIQUEMENT
--      (pas de rotation -> pas de bras tordus)
--   4) les associations valides sont conservées d'un refresh à l'autre
local function calibrerPaires(character, clone, anciennesPaires)
    local reelsNom = {}
    for _, part in ipairs(character:GetDescendants()) do
        if part:IsA("BasePart")
           and not part:FindFirstAncestorOfClass("Accessory") then
            reelsNom[part.Name] = part
        end
    end
    local reelsListe = {}
    for _, p in pairs(reelsNom) do
        table.insert(reelsListe, p)
    end

    local ancienneDe = {}
    if anciennesPaires then
        for _, pa in ipairs(anciennesPaires) do
            if pa.squelette and pa.squelette.Parent then
                ancienneDe[pa.squelette] = pa
            end
        end
    end

    -- parties squelette dans l'ordre hiérarchique (parent avant enfant)
    local partsSquelette = {}
    for _, part in ipairs(clone:GetDescendants()) do
        if part:IsA("BasePart")
           and not part:FindFirstAncestorOfClass("Accessory") then
            table.insert(partsSquelette, part)
        end
    end

    local paires = {}
    local pairDe = {} -- [partie squelette] = son association

    for _, part in ipairs(partsSquelette) do
        local ancienne = ancienneDe[part]
        local paire

        -- 0) ancienne association (nom/ancre) encore valide ?
        if ancienne and ancienne.reel and ancienne.reel.Parent
           and ancienne.reel:IsDescendantOf(character) then
            paire = ancienne

        else
            -- 1) correspondance par nom
            local reel = reelsNom[part.Name]
            if reel then
                paire = { squelette = part, reel = reel, offset = CFrame.new() }

            else
                local parentPart = part.Parent
                local parentValide = parentPart and parentPart:IsA("BasePart")
                    and not parentPart:FindFirstAncestorOfClass("Accessory")

                if parentValide and pairDe[parentPart] then
                    -- 2) CHAÎNE : suit son segment parent squelette
                    if ancienne and ancienne.suitPart == parentPart
                       and ancienne.offsetChaine then
                        paire = ancienne
                    else
                        paire = {
                            squelette = part,
                            suitPart = parentPart,
                            offsetChaine = parentPart.CFrame:Inverse() * part.CFrame,
                        }
                    end
                else
                    -- 3) racine de chaîne : ancre R6 canonique ou proximité
                    local ancre = (ANCRAGE_R6[part.Name]
                        and reelsNom[ANCRAGE_R6[part.Name]]) or nil
                    if not ancre then
                        local plusProche, dMin
                        for _, r in ipairs(reelsListe) do
                            local d = (r.Position - part.Position).Magnitude
                            if not dMin or d < dMin then
                                dMin, plusProche = d, r
                            end
                        end
                        ancre = plusProche
                    end
                    if ancre then
                        local rel = ancre.CFrame:Inverse() * part.CFrame
                        paire = {
                            squelette = part, reel = ancre,
                            offsetPos = rel.Position,
                        }
                    end
                end
            end
        end

        if paire then
            pairDe[part] = paire
            table.insert(paires, paire)
        end
    end
    return paires
end

-- Accessoires du squelette : suivent la partie du corps où ils sont fixés
local function calibrerAccessoires(clone)
    local liste = {}
    for _, acc in ipairs(clone:GetChildren()) do
        if acc:IsA("Accessory") then
            local handle = acc:FindFirstChild("Handle")
            local porte
            for _, w in ipairs(clone:GetDescendants()) do
                if (w:IsA("Weld") or w:IsA("WeldConstraint"))
                   and w.Part0 and w.Part1 then
                    if w.Part0 == handle then
                        porte = w.Part1
                    elseif w.Part1 == handle then
                        porte = w.Part0
                    end
                end
            end
            if handle and porte and porte ~= handle then
                table.insert(liste, {
                    handle = handle, porte = porte,
                    offset = porte.CFrame:Inverse() * handle.CFrame,
                })
            end
        end
    end
    return liste
end

-- Le Humanoid n'est PAS obligatoire (perso custom du jeu)
local function creerClone(player, character)
    local ancien = clones[player]
    if ancien and ancien.char == character and ancien.model
       and ancien.model.Parent then
        return
    end
    detruireClone(player)

    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local racine = trouverRacine(character)

    -- diagnostic : montre exactement la structure du perso du jeu
    print(("[Skin] ℹ️ Perso de %s : %d objets | Humanoid=%s | racine=%s")
        :format(player.Name, #character:GetDescendants(),
            tostring(humanoid ~= nil),
            racine and racine.Name or "AUCUNE"))

    local ok, clone = pcall(function()
        return Players:CreateHumanoidModelFromDescription(
            description:Clone(),
            humanoid and humanoid.RigType or Enum.HumanoidRigType.R15)
    end)
    if not ok or not clone then
        warn("[Skin] ⚠️ Création du clone échouée : " .. tostring(clone))
        return
    end

    clone.Name = "SkinSquelette"

    local hum2 = clone:FindFirstChildOfClass("Humanoid")
    if hum2 then hum2:Destroy() end
    local couleurInitiale = getCouleur()
    for _, s in ipairs(clone:GetDescendants()) do
        if s:IsA("BaseScript") or s:IsA("Motor6D") then
            s:Destroy()
        elseif s:IsA("BasePart") then
            s.Anchored = true
            s.CanCollide = false
            s.CastShadow = true
            s.Color = couleurInitiale
        elseif s:IsA("Decal") or s:IsA("Texture")
              or s:IsA("SurfaceAppearance") then
            s:Destroy()
        end
    end

    if racine then
        clone:PivotTo(racine.CFrame)
    end
    clone.Parent = workspace

    local cData = {
        model = clone,
        char = character,
        sync = nil,
        paires = {},
        accessoires = {},
    }

    local estMoi = (player == Players.LocalPlayer)
    local cam = workspace.CurrentCamera

    cData.sync = RunService.RenderStepped:Connect(function()
        if not clone.Parent or not character.Parent then
            cData.sync:Disconnect()
            clone:Destroy()
            if clones[player] and clones[player].model == clone then
                clones[player] = nil
            end
            return
        end

        for _, paire in ipairs(cData.paires) do
            local sq, reel = paire.squelette, paire.reel
            if sq.Parent then
                if paire.offset and reel and reel.Parent then
                    -- nom exact : copie directe de la partie réelle
                    sq.CFrame = reel.CFrame * paire.offset
                elseif paire.suitPart and paire.suitPart.Parent then
                    -- chaîne : suit son segment parent squelette
                    -- -> mains / pieds / segments TOUJOURS rattachés
                    sq.CFrame = paire.suitPart.CFrame * paire.offsetChaine
                elseif reel and reel.Parent and paire.offsetPos then
                    -- ancre : position seule, orientation du réel
                    sq.CFrame = reel.CFrame * CFrame.new(paire.offsetPos)
                end
            end
        end

        for _, paire in ipairs(cData.accessoires) do
            if paire.handle.Parent and paire.porte.Parent then
                paire.handle.CFrame = paire.porte.CFrame * paire.offset
            end
        end

        -- en vue première personne, cache ton propre squelette
        if estMoi and cam and racine and racine.Parent then
            local dist = (cam.CFrame.Position - racine.Position).Magnitude
            for _, paire in ipairs(cData.paires) do
                paire.squelette.LocalTransparencyModifier = (dist > 4) and 0 or 1
            end
            for _, paire in ipairs(cData.accessoires) do
                paire.handle.LocalTransparencyModifier = (dist > 4) and 0 or 1
            end
        end
    end)

    clones[player] = cData

    cData.paires = calibrerPaires(character, clone)
    cData.accessoires = calibrerAccessoires(clone)

    print("[Skin] 💀 Clone squelette créé pour " .. player.Name)
end

-- Refresh (manuel, touche P) : répare uniquement les associations mortes,
-- re-cache le vrai perso, remet la couleur
local function rafraichir(player)
    local c = clones[player]
    if not c or not c.model or not c.model.Parent then return end
    local character = c.char
    if not character or not character.Parent then return end

    c.paires = calibrerPaires(character, c.model, c.paires)
    c.accessoires = calibrerAccessoires(c.model)

    for _, obj in ipairs(character:GetDescendants()) do
        if not estDansUnTool(obj) then
            cacherObjet(obj)
        end
    end

    local couleur = getCouleur()
    for _, s in ipairs(c.model:GetDescendants()) do
        if s:IsA("BasePart") then
            s.Color = couleur
        end
    end
end

local function appliquer(player, character)
    if not character or not character.Parent then return end

    -- 1) ON SUPPRIME TON SKIN D'ABORD, toujours,
    --    même si le perso est custom (HumanoidRootPart absent etc.)
    masquerPerso(character)

    -- 2) puis on pose le squelette par-dessus
    creerClone(player, character)
end

-- Refresh manuel (touche P) : recalibre les skins existants,
-- ou les recrée s'ils ont disparu
local function rafraichirTout()
    local n = 0
    for _, player in ipairs(Players:GetPlayers()) do
        if estCible(player) and player.Character and player.Character.Parent then
            local c = clones[player]
            if c and c.model and c.model.Parent then
                pcall(rafraichir, player)
            else
                appliquer(player, player.Character)
            end
            n = n + 1
        end
    end
    print("[Skin] 🔄 Refresh manuel : " .. n .. " skin(s) recalibré(s) / réappliqué(s)")
end

local function surveiller(player)
    if not estCible(player) then return end

    local function onCharacter(char)
        -- le jeu habille le perso APRÈS le spawn -> quelques passages
        -- (l'habillage de départ reste auto ; le REFRESH, lui, est sur P)
        task.delay(1,  function() if char.Parent then appliquer(player, char) end end)
        task.delay(3,  function() if char.Parent then appliquer(player, char) end end)
        task.delay(6,  function() if char.Parent then appliquer(player, char) end end)
        task.delay(10, function() if char.Parent then appliquer(player, char) end end)
    end

    if player.Character then
        task.spawn(onCharacter, player.Character)
    end
    player.CharacterAdded:Connect(function(c)
        task.wait(0.5)
        onCharacter(c)
    end)
end

for _, p in ipairs(Players:GetPlayers()) do
    task.spawn(surveiller, p)
end
Players.PlayerAdded:Connect(function(p)
    task.spawn(surveiller, p)
end)

Players.PlayerRemoving:Connect(detruireClone)
